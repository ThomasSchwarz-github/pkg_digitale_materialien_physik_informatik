Die Klasse Kinosaal
/**
 * Ein Kinosaal, für den Karten verkauft werden.
 */
public class Kinosaal
{
    // Attribute
    private String film;
    private int plaetze;
    private int verkauft;
    private boolean geoeffnet;

    /**
     * Konstruktor: Ein neuer Saal hat 80 Plätze, noch keine
     * verkauften Karten, keinen Film und ist geschlossen.
     */
    public Kinosaal()
    {
        this.film = "kein Film";
        this.plaetze = 80;
        this.verkauft = 0;
        this.geoeffnet = false;
    }

    /**
     * Gibt den Titel des Films zurück.
     */
    public String gibFilm()
    {
        return this.film;
    }

    /**
     * Gibt die Anzahl der verkauften Karten zurück.
     */
    public int gibVerkauft()
    {
        return this.verkauft;
    }

    /**
     * Gibt zurück, ob der Saal geöffnet ist.
     */
    public boolean istGeoeffnet()
    {
        return this.geoeffnet;
    }

    /**
     * Setzt den Titel des Films auf den übergebenen Wert.
     */
    public void setzeFilm(String pFilm)
    {
        this.film = pFilm;
    }

    /**
     * Öffnet den Saal.
     */
    public void oeffne()
    {
        this.geoeffnet = true;
    }

    /**
     * Verkauft die übergebene Anzahl Karten.
     */
    public void verkaufe(int pAnzahl)
    {
        this.verkauft = this.verkauft + pAnzahl;
    }

    /**
     * Gibt zurück, wie viele Plätze noch frei sind.
     */
    public int berechneFreiePlaetze()
    {
        return this.plaetze - this.verkauft;
    }

    /**
     * Gibt die Einnahmen bei dem übergebenen Preis zurück.
     */
    public int berechneEinnahmen(int pPreis)
    {
        return this.verkauft * pPreis;
    }

    /**
     * Gibt die Auslastung in Prozent zurück.
     */
    public int berechneAuslastung()
    {
        return this.verkauft * 100 / this.plaetze;
    }
}
