Die Klasse Lautsprecher
/**
 * Eine Bluetooth-Box mit Name, Lautstärke und Akku.
 */
public class Lautsprecher
{
    // Attribute
    private String name;
    private int lautstaerke;
    private int akkustand;
    private boolean eingeschaltet;

    /**
     * Konstruktor: Ein neuer Lautsprecher heißt "Box",
     * hat Lautstärke 10, einen vollen Akku und ist aus.
     */
    public Lautsprecher()
    {
        this.name = "Box";
        this.lautstaerke = 10;
        this.akkustand = 100;
        this.eingeschaltet = false;
    }

    /**
     * Schaltet den Lautsprecher ein.
     */
    public void schalteEin()
    {
        this.eingeschaltet = true;
    }

    /**
     * Setzt den Namen auf den übergebenen Wert.
     */
    public void setzeName(String pName)
    {
        this.name = pName;
    }

    /**
     * Setzt die Lautstärke auf den übergebenen Wert.
     */
    public void setzeLautstaerke(int pLautstaerke)
    {
        this.lautstaerke = pLautstaerke;
    }

    /**
     * Erhöht die Lautstärke um die übergebenen Stufen.
     */
    public void lauterUm(int pStufen)
    {
        this.lautstaerke = this.lautstaerke + pStufen;
    }

    /**
     * Spielt ein Lied. Das kostet 5 Prozent Akku.
     */
    public void spieleLied()
    {
        this.akkustand = this.akkustand - 5;
        System.out.println(this.name + " spielt ein Lied.");
    }
}
